------ Start Spark Sttreaming 

docker exec -it spark-master `
  /opt/spark/bin/spark-submit `
  --master spark://spark-master:7077 `
  --conf spark.jars.ivy=/tmp/.ivy2 `
  --packages org.apache.spark:spark-sql-kafka-0-10_2.12:3.5.5,com.datastax.spark:spark-cassandra-connector_2.12:3.5.1 `
  --conf spark.cassandra.connection.host=cassandra `
  --conf spark.cassandra.connection.port=9042 `
  /opt/spark/work-dir/spark_stream.py


  -------verify cassandra 

  docker exec cassandra cqlsh -e "SELECT COUNT(*) FROM spark_streams.created_users;"

  docker exec cassandra cqlsh -e "SELECT first_name, last_name, email, username FROM spark_streams.created_users LIMIT 10;"
